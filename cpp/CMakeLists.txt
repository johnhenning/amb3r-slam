cmake_minimum_required(VERSION 3.20)
project(amber_inference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime distribution with include/ and lib/")
find_path(ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" REQUIRED)
find_library(ORT_LIBRARY onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
add_library(amber_inference STATIC inference.cpp)
target_include_directories(amber_inference PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${ORT_INCLUDE})
target_link_libraries(amber_inference PUBLIC ${ORT_LIBRARY})
add_executable(amber_infer main.cpp)
target_link_libraries(amber_infer PRIVATE amber_inference)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(amber_inference PRIVATE -Wall -Wextra -Wpedantic)
endif()
